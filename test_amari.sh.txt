node <<'NODE'
const WebSocket = require('ws');
const { randomUUID } = require('node:crypto');

const messageId = `manual-test#${randomUUID()}`;
const ws = new WebSocket(process.env.GNB_URL, {
  origin: process.env.GNB_ORIGIN,
  handshakeTimeout: 5000,
});

let commandSent = false;

const timeout = setTimeout(() => {
  console.error('FAILED: timed out waiting for Amarisoft response');
  ws.terminate();
  process.exitCode = 1;
}, 10000);

function sendUeGet() {
  if (commandSent) return;
  commandSent = true;

  const request = {
    message: 'ue_get',
    message_id: messageId,
    stats: true,
  };

  console.log('Sending ue_get');
  ws.send(JSON.stringify(request));
}

ws.on('open', () => {
  console.log('WebSocket connected');
});

ws.on('message', (data) => {
  let response;

  try {
    response = JSON.parse(data.toString());
  } catch {
    console.error('FAILED: received non-JSON response');
    return;
  }

  if (response.message === 'ready') {
    console.log(
      `Amarisoft ready: ${response.name ?? 'unknown'} ` +
      `${response.type ?? ''} ${response.version ?? ''}`,
    );
    sendUeGet();
    return;
  }

  if (response.message === 'authenticate') {
    console.error('Authentication is required; configure the Amarisoft password');
    clearTimeout(timeout);
    ws.close();
    process.exitCode = 2;
    return;
  }

  if (response.message_id === messageId) {
    console.log('SUCCESS: ue_get response received');
    console.log('Response fields:', Object.keys(response).join(', '));

    const ueList = response.ue_list ?? response.ues;
    if (Array.isArray(ueList)) {
      console.log('UE count:', ueList.length);
    }

    clearTimeout(timeout);
    ws.close();
  }
});

ws.on('unexpected-response', (_request, response) => {
  console.error(
    'FAILED HTTP response:',
    response.statusCode,
    response.statusMessage,
  );
  response.resume();
  clearTimeout(timeout);
  process.exitCode = 1;
});

ws.on('error', (error) => {
  console.error('FAILED:', error.message);
  clearTimeout(timeout);
  process.exitCode = 1;
});

ws.on('close', (code, reason) => {
  console.log('Connection closed:', code, reason.toString());
});
NODE
